1. Story Generation (Groq API)

import requests url = "https://api.groq.com/openai/v1/chat/completions" headers = {"Authorization": "Bearer YOUR_API_KEY"} data = { "model": "llama-3.1-8b-instant", "messages": [{"role": "user", "content": "Write a short story about a robot learning emotions."}] } response = requests.post(url, json=data, headers=headers) print(response.json()["choices"][0]["message"]["content"])

2. Image Generation (Groq API)

import requests url = "https://api.groq.com/openai/v1/images/generations" headers = {"Authorization": "Bearer YOUR_API_KEY"} data = { "model": "whisperless-image-1", "prompt": "A futuristic city floating in the clouds" } response = requests.post(url, json=data, headers=headers) print(response.json())
