1️⃣ Story Generation (Python + Groq API)


import requests

api_key = "YOUR_GROQ_API_KEY"

url = "https://api.groq.com/openai/v1/chat/completions"

prompt = "Write a short story about a lost puppy that finds a new home."

data = {
    "model": "llama-3.1-8b-instant",
    "messages": [
        {"role": "user", "content": prompt}
    ]
}

headers = {
    "Authorization": f"Bearer {api_key}",
    "Content-Type": "application/json"
}

response = requests.post(url, json=data, headers=headers)
result = response.json()

print("\n--- Story Generated ---\n")
print(result["choices"][0]["message"]["content"])

2️⃣ Image Generation (Stability AI - Python)


import requests

API_KEY = "YOUR_STABILITY_API_KEY"
ENDPOINT = "https://api.stability.ai/v2beta/stable-image/generate/sd3"

def generate_image(prompt):
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Accept": "image/*"
    }
    files = {
        "prompt": (None, prompt),
        "output_format": (None, "png")
    }

    response = requests.post(ENDPOINT, headers=headers, files=files)

    if response.status_code == 200:
        with open("output.png", "wb") as f:
            f.write(response.content)
        print("Image saved as output.png")
    else:
        print("Error:", response.status_code, response.text)

prompt = input("Enter image description: ")
generate_image(prompt)

3️⃣ Streamlit Story Generator (Simple)


import requests
import streamlit as st

st.title('Simple Story Generator')
prompt = st.text_input('Enter a story idea:')

api_key = "YOUR_GROQ_API_KEY"
url = "https://api.groq.com/openai/v1/chat/completions"

if st.button('Generate Story'):
    if prompt.strip() == "":
        st.warning('Please enter a story idea!')
    else:
        data = {
            "model": "llama-3.1-8b-instant",
            "messages": [{"role": "user", "content": prompt}]
        }
        headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        }
        response = requests.post(url, json=data, headers=headers)
        result = response.json()

        story = result["choices"][0]["message"]["content"]

        st.subheader('Generated Story')
        st.write(story)

5️⃣ Streamlit – Skin Analyzer (Groq)


import streamlit as st
import base64
import requests
from PIL import Image
import io

st.title("Skin Analyzer (Groq Free Tier)")

api_key = st.text_input("Enter Groq API Key", type="password")
uploaded_file = st.file_uploader("Upload your face image", type=["jpg", "jpeg", "png"])

def ultra_compress(image_bytes, max_size=256, quality=20):
    img = Image.open(io.BytesIO(image_bytes)).convert("RGB")
    img.thumbnail((max_size, max_size))
    buffer = io.BytesIO()
    img.save(buffer, format="JPEG", quality=quality)
    return buffer.getvalue()

if uploaded_file and api_key:
    compressed = ultra_compress(uploaded_file.read())
    img_base64 = base64.b64encode(compressed).decode()

    user_msg = (
        "Analyze my skin.\n"
        "Image base64 below:\n"
        f"{img_base64}"
    )

    payload = {
        "model": "llama-3.1-8b-instant",
        "messages": [{"role": "user", "content": user_msg}]
    }

    headers = {
        "Authorization": f"Bearer {api_key}",
        "Content-Type": "application/json"
    }

    response = requests.post(
        "https://api.groq.com/openai/v1/chat/completions",
        json=payload, headers=headers
    )

    try:
        st.write(response.json()["choices"][0]["message"]["content"])
    except:
        st.error(response.json())
