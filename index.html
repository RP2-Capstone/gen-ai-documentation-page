1️⃣ Story Generation (Python + Groq API)


import requests

api_key = "YOUR_GROQ_API_KEY"

url = "https://api.groq.com/openai/v1/chat/completions"

prompt = "Write a short story about a lost puppy that finds a new home."

data = {
    "model": "llama-3.1-8b-instant",
    "messages": [
        {"role": "user", "content": prompt}
    ]
}

headers = {
    "Authorization": f"Bearer {api_key}",
    "Content-Type": "application/json"
}

response = requests.post(url, json=data, headers=headers)
result = response.json()

print("\n--- Story Generated ---\n")
print(result["choices"][0]["message"]["content"])

2️⃣ Image Generation (Stability AI - Python)


import requests

API_KEY = "YOUR_STABILITY_API_KEY"
ENDPOINT = "https://api.stability.ai/v2beta/stable-image/generate/sd3"

def generate_image(prompt):
    headers = {
        "Authorization": f"Bearer {API_KEY}",
        "Accept": "image/*"
    }
    files = {
        "prompt": (None, prompt),
        "output_format": (None, "png")
    }

    response = requests.post(ENDPOINT, headers=headers, files=files)

    if response.status_code == 200:
        with open("output.png", "wb") as f:
            f.write(response.content)
        print("Image saved as output.png")
    else:
        print("Error:", response.status_code, response.text)

prompt = input("Enter image description: ")
generate_image(prompt)

3️⃣ Streamlit Story Generator (Simple)


import requests
import streamlit as st

st.title('Simple Story Generator')
prompt = st.text_input('Enter a story idea:')

api_key = "YOUR_GROQ_API_KEY"
url = "https://api.groq.com/openai/v1/chat/completions"

if st.button('Generate Story'):
    if prompt.strip() == "":
        st.warning('Please enter a story idea!')
    else:
        data = {
            "model": "llama-3.1-8b-instant",
            "messages": [{"role": "user", "content": prompt}]
        }
        headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        }
        response = requests.post(url, json=data, headers=headers)
        result = response.json()

        story = result["choices"][0]["message"]["content"]

        st.subheader('Generated Story')
        st.write(story)
